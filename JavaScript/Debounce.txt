Question : 

You have this HTML:

<input id="search" />

<div id="results"> </div>

Write a debounce wrapper and show how you'd use it so that a function doSearch(term) runs only if the user has stopped typing for 500ms.
Provide the debounce implementation and the event listener code (no need to implement doSearch).
function doSearch(){
    Console.log (“searching”)
}
function debounce(){
    // implement debounce
}


Answer : 

<input id="search" />

<div id="results"> </div>
 
<script>
    // Example search function
    function doSearch(term) {
        console.log("searching:", term);
    }

    // --- Debounce implementation ---
    function debounce(fn, delay) {
        let timer;
        return function (...args) {
            clearTimeout(timer);
            timer = setTimeout(() => fn.apply(this, args), delay);
        };
    }

    // --- Usage with the input field ---
    const input = document.getElementById("search");

    // Wrap doSearch in a debounced version that triggers after 500ms
    const debouncedSearch = debounce(doSearch, 500);

    input.addEventListener("input", (e) => {
        debouncedSearch(e.target.value);
    });
</script>

______________________________________________________________________________________________________________________________________

Debouncing means delaying a function’s execution until the user stops triggering it for a specified amount of time.
